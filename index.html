// Optimización de búsqueda binaria
function binarySearch<T>(
  arr: T[], 
  target: T, 
  compare: (a: T, b: T) => number
): number {
  let left = 0, right = arr.length - 1;
  
  while (left <= right) {
    const mid = (left + right) >>> 1;
    const cmp = compare(arr[mid], target);
    
    if (cmp === 0) return mid;
    if (cmp < 0) left = mid + 1;
    else right = mid - 1;
  }
  return -1; // No encontrado
}
